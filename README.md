   7. SAILOR, RESERVES and BOATS – INTERSECT, UNION ALL, EXISTS
CREATE TABLE SAILOR(SID NUMBER(5), SNAME VARCHAR2(30), RATING NUMBER(3), AGE NUMBER(5,1));
CREATE TABLE BOATS(BID NUMBER(5), BNAME VARCHAR2(30), COLOR VARCHAR2(20));
CREATE TABLE RESERVES(SID NUMBER(5), BID NUMBER(5), DAY DATE);
INSERT INTO BOATS VALUES(101,'Interlake','blue');
INSERT INTO BOATS VALUES(102,'Interlake','red');
INSERT INTO BOATS VALUES(103,'Clipper','green');
INSERT INTO BOATS VALUES(104,'Marine','red');
-- Sailors who reserved both red and green boats:
SELECT SNAME FROM SAILOR WHERE SID IN
 (SELECT R.SID FROM RESERVES R JOIN BOATS B ON R.BID=B.BID WHERE B.COLOR='red')
INTERSECT
SELECT SNAME FROM SAILOR WHERE SID IN
 (SELECT R.SID FROM RESERVES R JOIN BOATS B ON R.BID=B.BID WHERE B.COLOR='green');
-- Equivalent both-color test using GROUP BY:
SELECT S.SNAME FROM SAILOR S JOIN RESERVES R ON S.SID=R.SID
JOIN BOATS B ON R.BID=B.BID WHERE B.COLOR IN ('red','green')
GROUP BY S.SID,S.SNAME HAVING COUNT(DISTINCT B.COLOR)=2;
-- Sailors who reserved boat 103:
SELECT SNAME FROM SAILOR S WHERE EXISTS
 (SELECT * FROM RESERVES R WHERE R.SID=S.SID AND R.BID=103);
The paper labels part 7(b) UNION ALL, but UNION ALL alone means combine rows including duplicates; it does not mean both
colors. The GROUP BY query above correctly tests for both.

    9. CUSTOMER and ORDER_TABLE joins
CREATE TABLE CUSTOMER (
 ID NUMBER(5) PRIMARY KEY, NAME VARCHAR2(30), AGE NUMBER(3),
 ADDRESS VARCHAR2(30), SALARY NUMBER(10,2)
);
CREATE TABLE ORDER_TABLE (OID NUMBER(5), ODAY DATE, CUSTOMER_ID NUMBER(5), AMOUNT NUMBER(10,2));
-- INNER JOIN
SELECT C.ID,C.NAME,O.OID,O.AMOUNT FROM CUSTOMER C
INNER JOIN ORDER_TABLE O ON C.ID=O.CUSTOMER_ID;
-- LEFT OUTER JOIN
SELECT C.ID,C.NAME,O.OID,O.AMOUNT FROM CUSTOMER C
LEFT OUTER JOIN ORDER_TABLE O ON C.ID=O.CUSTOMER_ID;
-- RIGHT OUTER JOIN
SELECT C.ID,C.NAME,O.OID,O.AMOUNT FROM CUSTOMER C
RIGHT OUTER JOIN ORDER_TABLE O ON C.ID=O.CUSTOMER_ID;

-- Rating greater than at least one rating of Horatio:

WHERE RATING > ANY (SELECT RATING FROM SAILOR WHERE SNAME='Horatio');
SELECT SNAME,RATING FROM SAILOR

-- Sailors who reserved a red OR green boat:
SELECT S.SNAME FROM SAILOR S JOIN RESERVES R ON S.SID=R.SID
JOIN BOATS B ON R.BID=B.BID WHERE B.COLOR='red'
UNION
SELECT S.SNAME FROM SAILOR S JOIN RESERVES R ON S.SID=R.SID
JOIN BOATS B ON R.BID=B.BID WHERE B.COLOR='green';
UNION in 11(c) means red OR green. For both colors, use INTERSECT or the GROUP BY/HAVING query in Question 7.

 ACTION_TYPE VARCHAR2(20), ACTION_DATE DATE);

AFTER INSERT OR UPDATE OR DELETE ON EMPLOYEE
FOR EACH ROW
BEGIN
 IF INSERTING THEN
 INSERT INTO EMP_AUDIT VALUES(:NEW.EMPNO,:NEW.ENAME,'INSERT',SYSDATE);
 ELSIF UPDATING THEN
 INSERT INTO EMP_AUDIT VALUES(:NEW.EMPNO,:NEW.ENAME,'UPDATE',SYSDATE);
 ELSIF DELETING THEN
 INSERT INTO EMP_AUDIT VALUES(:OLD.EMPNO,:OLD.ENAME,'DELETE',SYSDATE);
 END IF;
END;

13(b) Normalization
1NF: Every field contains one atomic value; no list of values in a single cell.
-- 1NF example
SID SNAME COURSE
1 Ravi SQL
1 Ravi Python
2 Kiran Java
2NF: Table is in 1NF and every non-key attribute depends on the whole candidate key; no partial dependency.
STUDENT(SID, SNAME)
SID SNAME
1 Ravi
2 Kiran
STUDENT_COURSE(SID, COURSE)
SID COURSE
1 SQL
1 Python
2 Java
3NF: Table is in 2NF and has no transitive dependency of non-key attributes on a key.
STUDENT(SID, SNAME, DEPT_ID)
SID SNAME DEPT_ID
1 Ravi 10
2 Kiran 20
DEPARTMENT(DEPT_ID, DEPT_NAME)
DEPT_ID DEPT_NAME
10 CSE
20 ECE
